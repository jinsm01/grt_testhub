## 关键代码片段

### 1. 后端上传 API

```python
@api_view(['POST'])
@permission_classes([IsAuthenticated])
def upload_temp_file(request):
    """上传临时文件到临时目录"""
    uploaded_file = request.FILES['file']
    scenario_id = request.data.get('scenario_id')
    
    # 生成唯一文件标识
    file_id = str(uuid.uuid4())
    
    # 创建临时文件目录
    temp_dir = os.path.join(settings.MEDIA_ROOT, 'temp_files')
    if scenario_id:
        temp_dir = os.path.join(temp_dir, f'scenario_{scenario_id}')
    
    # 保存文件
    file_path = os.path.join(temp_dir, f"{file_id[:8]}_{uploaded_file.name}")
    
    # 返回文件信息
    return Response({
        'file_id': file_id,
        'file_path': file_path.replace(settings.MEDIA_ROOT + os.sep, ''),
        'file_name': uploaded_file.name,
        'file_size': uploaded_file.size,
        'content_type': uploaded_file.content_type
    })
```

### 2. 执行引擎文件处理

```python
if body_type == 'form-data':
    multipart_data = {}
    has_files = False
    
    for item in body_data:
        if item['type'] == 'file':
            file_path = item.get('file_path')
            if file_path:
                # 从临时目录读取文件
                full_file_path = os.path.join(settings.MEDIA_ROOT, file_path)
                if os.path.exists(full_file_path):
                    with open(full_file_path, 'rb') as f:
                        file_content = f.read()
                    
                    multipart_data[item['key']] = (
                        item['value'],  # 原始文件名
                        file_content,
                        item.get('content_type', 'application/octet-stream')
                    )
                    has_files = True
    
    if has_files:
        encoder = MultipartEncoder(fields=multipart_data)
        headers_dict['Content-Type'] = encoder.content_type
        request_kwargs['data'] = encoder
```

### 3. 前端文件上传

```javascript
const handleFileChange = async (index, uploadFile) => {
  // 创建 FormData 上传文件
  const formData = new FormData()
  formData.append('file', uploadFile.raw)
  if (currentScenarioId.value) {
    formData.append('scenario_id', currentScenarioId.value)
  }
  
  // 上传文件到临时目录
  const response = await axios.post('/api-testing/temp-files/upload/', formData)
  
  // ✨ 保存返回的文件信息
  rows.value[index].file_path = response.data.file_path  // 关键字段
  rows.value[index].value = response.data.file_name
  rows.value[index].file_size = response.data.file_size
  rows.value[index].content_type = response.data.content_type
  
  updateValue()
}
```

      "enabled": true,

  ]
}
